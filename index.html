function compileExcelAndMerge(matrix) { if(!matrix || matrix.length < 3) return; const rowDays = matrix[0]; const rowDates = matrix[1]; let newIncomingDays = []; TARGET_EXCEL_COLUMNS.forEach(col => { let actualColIdx = col.idx; if (actualColIdx >= rowDays.length) return; let rawDayVal = rowDays[actualColIdx]; let rawDateVal = rowDates[actualColIdx]; if (rawDateVal || rawDayVal) { let parsedDate = parseExcelDate(rawDateVal); let formatted = parsedDate ? parsedDate.formatted : ""; let mm = parsedDate ? parsedDate.mm : "06"; let yy = parsedDate ? parsedDate.yy : "2026"; let computedDayTitle = rawDayVal ? String(rawDayVal).trim().toUpperCase() : ""; if (!computedDayTitle && formatted) { let p = (formatted || "").split('.'); if(p.length === 3) { let calendarCheck = new Date(p[2], p[1] - 1, p[0]); computedDayTitle = TURKISH_DAYS[calendarCheck.getDay()].toUpperCase(); } } if (formatted) { newIncomingDays.push({ dateTitle: formatted, dayTitle: computedDayTitle, month: mm, year: yy, excelColIdx: actualColIdx }); } } }); newIncomingDays.sort((a, b) => { let pA = (a.dateTitle || "").split('.'), pB = (b.dateTitle || "").split('.'); if(pA.length === 3 && pB.length === 3) { return new Date(`${pA[2]}-${pA[1]}-${pA[0]}`) - new Date(`${pB[2]}-${pB[1]}-${pB[0]}`); } return 0; }); timePeriodDaysList = [...newIncomingDays]; const newDatesSet = newIncomingDays.map(d => d.dateTitle); mainShiftsLogPool = mainShiftsLogPool.filter(log => !newDatesSet.includes(log.date) || log.uniqueKey.startsWith("leave_")); newIncomingDays.forEach(dayColumn => { let actualColIdx = dayColumn.excelColIdx; let blockStructureList = []; let currentBlock = null; for (let rIdx = 2; rIdx < matrix.length; rIdx++) { if (!matrix[rIdx]) continue; let cellVal = matrix[rIdx][actualColIdx]; let cellHour = matrix[rIdx][actualColIdx + 2]; let cleanStr = cellVal ? String(cellVal).trim() : ""; if (cleanStr === "null" || cleanStr === "") continue; let sanitizedCheck = cleanStr.replace(/\.+$/, "").trim(); let isPerson = USER_EXPLICIT_25.includes(sanitizedCheck); // EĞER Hücre Personel İsim Listesinde Değilse, Bu Yeni Bir İŞ BAŞLIĞIDIR (Örn: GENÇ NEFES, FUTBOL vb.) if (!isPerson) { currentBlock = { programName: cleanStr, hour: cellHour ? String(cellHour).trim() : "Belirtilmedi", rowStart: rIdx, uniqueKey: dayColumn.dateTitle + "_" + actualColIdx + "_b_" + rIdx, subLines: [] }; blockStructureList.push(currentBlock); } else if (currentBlock) { // Eğer hücre personelse, en son açılmış olan kalın çerçeveli iş bloğuna eklenir currentBlock.subLines.push({ text: cleanStr, rowIdx: rIdx }); } } blockStructureList.forEach(block => { let items = block.subLines; for (let i = 0; i < items.length; i++) { let currentText = items[i].text; let hasDot = currentText.endsWith("."); let sanitizedName = currentText.replace(/\.+$/, "").trim(); if (USER_EXPLICIT_25.includes(sanitizedName)) { let descriptionLines = []; for (let j = i - 1; j >= 0; j--) { let prevText = items[j].text; let prevSanitized = prevText.replace(/\.+$/, "").trim(); if (USER_EXPLICIT_25.includes(prevSanitized)) break; descriptionLines.unshift(prevText); } for (let j = i + 1; j < items.length; j++) { let nextText = items[j].text; let nextSanitized = nextText.replace(/\.+$/, "").trim(); if (USER_EXPLICIT_25.includes(nextSanitized)) break; descriptionLines.push(nextText); } mainShiftsLogPool.push({ workerName: sanitizedName, date: dayColumn.dateTitle, dayName: dayColumn.dayTitle, month: dayColumn.month, year: dayColumn.year, program: block.programName, subDetail: descriptionLines.join(" / "), hour: block.hour, uniqueKey: block.uniqueKey, isProbability: hasDot }); } } }); }); staffLeavePool.forEach(leave => { let currentDate = new Date(leave.start); const stopDate = new Date(leave.end); while (currentDate <= stopDate) { let dayNum = currentDate.getDate(); let monthNum = currentDate.getMonth() + 1; let yearNum = currentDate.getFullYear(); let formattedDate = `${dayNum < 10 ? '0'+dayNum : dayNum}.${monthNum < 10 ? '0'+monthNum : monthNum}.${yearNum}`; let computedDayTitle = TURKISH_DAYS[currentDate.getDay()].toUpperCase(); let checkKey = "leave_" + leave.id + "_" + formattedDate; if (!mainShiftsLogPool.some(l => l.uniqueKey === checkKey)) { mainShiftsLogPool.push({ workerName: leave.workerName, date: formattedDate, dayName: computedDayTitle, month: (monthNum < 10 ? '0'+monthNum : monthNum).toString(), year: yearNum.toString(), program: "YILLIK İZİN GÖREVİ", subDetail: "Ortak Sistem Tarafından İzinli İşlendi", hour: "00:00", uniqueKey: checkKey, isProbability: false }); } currentDate.setDate(currentDate.getDate() + 1); } }); if (newIncomingDays.length > 0) { let p = (newIncomingDays[0].dateTitle || "").split('.'); if(p.length === 3) document.getElementById('analysis-start-date').value = `${p[2]}-${p[1]}-${p[0]}`; let lastP = (newIncomingDays[newIncomingDays.length - 1].dateTitle || "").split('.'); if(lastP.length === 3) document.getElementById('analysis-end-date').value = `${lastP[2]}-${lastP[1]}-${lastP[0]}`; } saveDatabaseToLocalStorage(); rebuildStaffDropdownOptions(); }
