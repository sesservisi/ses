function compileExcelAndMerge(matrix) { if(!matrix || matrix.length < 3) return; const rowDays = matrix[0]; const rowDates = matrix[1]; let newIncomingDays = []; // 1. Excel'deki hedef sütunların ve tarihlerin tespiti TARGET_EXCEL_COLUMNS.forEach(col => { let actualColIdx = col.idx; if (actualColIdx >= rowDays.length) return; let rawDayVal = rowDays[actualColIdx]; let rawDateVal = rowDates[actualColIdx]; if (rawDateVal || rawDayVal) { let parsedDate = parseExcelDate(rawDateVal); let formatted = parsedDate ? parsedDate.formatted : ""; let mm = parsedDate ? parsedDate.mm : "08"; let yy = parsedDate ? parsedDate.yy : "2026"; let computedDayTitle = rawDayVal ? String(rawDayVal).trim().toUpperCase() : ""; if (!computedDayTitle && formatted) { let p = (formatted || "").split('.'); if(p.length === 3) { let calendarCheck = new Date(p[2], p[1] - 1, p[0]); computedDayTitle = TURKISH_DAYS[calendarCheck.getDay()].toUpperCase(); } } if (formatted) { newIncomingDays.push({ dateTitle: formatted, dayTitle: computedDayTitle, month: mm, year: yy, excelColIdx: actualColIdx }); } } }); // Tarihleri kronolojik sıraya dizme newIncomingDays.sort((a, b) => { let pA = (a.dateTitle || "").split('.'), pB = (b.dateTitle || "").split('.'); if(pA.length === 3 && pB.length === 3) { return new Date(`${pA[2]}-${pA[1]}-${pA[0]}`) - new Date(`${pB[2]}-${pB[1]}-${pB[0]}`); } return 0; }); timePeriodDaysList = [...newIncomingDays]; const newDatesSet = newIncomingDays.map(d => d.dateTitle); // Yıllık izin dışındaki eski kayıtları temizleme (İzin kayıplarını önler) mainShiftsLogPool = mainShiftsLogPool.filter(log => !newDatesSet.includes(log.date) || log.uniqueKey.startsWith("leave_")); // 2. Çerçeve/Kutu Tabanlı İş ve Personel Ayrıştırma newIncomingDays.forEach(dayColumn => { let actualColIdx = dayColumn.excelColIdx; let currentBlock = null; // Çerçeveli kutu takip nesnesi for (let rIdx = 2; rIdx < matrix.length; rIdx++) { if (!matrix[rIdx]) continue; let cellVal = matrix[rIdx][actualColIdx]; let cellHour = matrix[rIdx][actualColIdx + 2]; let cleanStr = cellVal ? String(cellVal).trim() : ""; // Boş satır (kutular arası boşluk) görüldüğünde mevcut kutuyu kapatıyoruz if (cleanStr === "null" || cleanStr === "") { if (currentBlock && currentBlock.staffList.length > 0) { currentBlock = null; // Kutu başarıyla sonlandırıldı } continue; } // Personel kontrolü let sanitizedName = cleanStr.replace(/\.+$/, "").trim(); let isStaff = USER_EXPLICIT_25.includes(sanitizedName); if (!isStaff) { // Eğer hücre personel değilse yeni bir iş kutusunun başlığı veya detay satırıdır if (!currentBlock || currentBlock.staffList.length > 0) { currentBlock = { title: cleanStr, subDetails: [], hour: (cellHour && String(cellHour).trim() !== "null") ? String(cellHour).trim() : "-", staffList: [] }; } else { // İş başlığının altına eklenen detaylar (Örn: Sinan Erdem, Sarıyer, KORUMA) currentBlock.subDetails.push(cleanStr); if (cellHour && currentBlock.hour === "-" && String(cellHour).trim() !== "null") { currentBlock.hour = String(cellHour).trim(); } } } else { // Hücre personel ismi ise, o an açık olan kutunun altına ekle if (currentBlock) { let isNoktaliGri = cleanStr.endsWith("."); let uniqueJobKey = dayColumn.dateTitle + "_" + actualColIdx + "_" + currentBlock.title + "_" + rIdx; mainShiftsLogPool.push({ workerName: sanitizedName, date: dayColumn.dateTitle, dayName: dayColumn.dayTitle, month: dayColumn.month, year: dayColumn.year, program: currentBlock.title, subDetail: currentBlock.subDetails.join(" / "), hour: currentBlock.hour, uniqueKey: uniqueJobKey, isProbability: isNoktaliGri // Noktalı/gri personel ayırıcı }); currentBlock.staffList.push(sanitizedName); } } } }); // 3. Yıllık İzinlerin Sistemle Senkronize Edilmesi staffLeavePool.forEach(leave => { let currentDate = new Date(leave.start); const stopDate = new Date(leave.end); while (currentDate <= stopDate) { let dayNum = currentDate.getDate(); let monthNum = currentDate.getMonth() + 1; let yearNum = currentDate.getFullYear(); let formattedDate = `${dayNum < 10 ? '0'+dayNum : dayNum}.${monthNum < 10 ? '0'+monthNum : monthNum}.${yearNum}`; let computedDayTitle = TURKISH_DAYS[currentDate.getDay()].toUpperCase(); let checkKey = "leave_" + leave.id + "_" + formattedDate; if (!mainShiftsLogPool.some(l => l.uniqueKey === checkKey)) { mainShiftsLogPool.push({ workerName: leave.workerName, date: formattedDate, dayName: computedDayTitle, month: (monthNum < 10 ? '0'+monthNum : monthNum).toString(), year: yearNum.toString(), program: "YILLIK İZİN GÖREVİ", subDetail: "Ortak Sistem Tarafından İzinli İşlendi", hour: "00:00", uniqueKey: checkKey, isProbability: false }); } currentDate.setDate(currentDate.getDate() + 1); } }); // 4. Analiz Tarih Aralıklarının Güncellenmesi ve Kayıt if (newIncomingDays.length > 0) { let p = (newIncomingDays[0].dateTitle || "").split('.'); if(p.length === 3) document.getElementById('analysis-start-date').value = `${p[2]}-${p[1]}-${p[0]}`; let lastP = (newIncomingDays[newIncomingDays.length - 1].dateTitle || "").split('.'); if(lastP.length === 3) document.getElementById('analysis-end-date').value = `${lastP[2]}-${lastP[1]}-${lastP[0]}`; } saveDatabaseToLocalStorage(); rebuildStaffDropdownOptions(); }
